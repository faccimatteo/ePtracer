cmake_minimum_required(VERSION 3.10)

project(eptracer)

# Set the C compiler to clang
set(CMAKE_C_COMPILER clang)
# Set the output directory for the compiled binaries
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

# Defining useful variables
set(PROJECT_SOURCE_DIR ${CMAKE_SOURCE_DIR}/src)
set(PROJECT_BUILD_DIR ${CMAKE_SOURCE_DIR}/build)
set(LOG_SRC_DIR ${CMAKE_SOURCE_DIR}/log.c/src)
set(OUTPUT_DIR ${CMAKE_SOURCE_DIR}/.output)

# Get the architecture
execute_process(
    COMMAND uname -m
    OUTPUT_VARIABLE ARCH
    OUTPUT_STRIP_TRAILING_WHITESPACE
)

# Transform the architecture string
if(ARCH STREQUAL "x86_64")
    set(ARCH "x86")
elseif(ARCH MATCHES "arm.*")
    set(ARCH "arm")
elseif(ARCH STREQUAL "aarch64")
    set(ARCH "arm64")
elseif(ARCH STREQUAL "ppc64le")
    set(ARCH "powerpc")
elseif(ARCH MATCHES "mips.*")
    set(ARCH "mips")
elseif(ARCH STREQUAL "riscv64")
    set(ARCH "riscv")
elseif(ARCH STREQUAL "loongarch64")
    set(ARCH "loongarch")
endif()
# Building shared library from log repository
# add_library(log.so SHARED EXCLUDE_FROM_ALL ${LOG_SRC_DIR}/log.c)

# Building BPF 
add_custom_target(eptracer_bpf_o ALL		        
	COMMAND ${CMAKE_C_COMPILER} -target bpf -I /usr/include/${CMAKE_SYSTEM_PROCESSOR}-linux-gnu -g -O2 -c ${PROJECT_SOURCE_DIR}/eptracer.bpf.c -o ${CMAKE_BINARY_DIR}/eptracer.bpf.o
	COMMENT "Building BPF program"
)

# Building user space 
add_custom_target(eptracer_o ALL		        
	COMMAND ${CMAKE_C_COMPILER} -g3 -O2 -c ${PROJECT_SOURCE_DIR}/eptracer.c -o ${CMAKE_BINARY_DIR}/eptracer.o
	DEPENDS bpf_skeleton
	COMMENT "Building user space program"
)
